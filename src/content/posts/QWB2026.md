### part3 算法还原

30 轮 Feistel 结构 

```Plain Text
K[0] = u64(0x7686) ^ u64(0x768e)
     = 0x1fa6bee7224fb586
K[1] = u64(0x7696) ^ u64(0x769e)
     = 0x56fc5596cf5bad58
原始字节：d7d2a74fa835b34f9ff4f4b4e383a102
目标左半：0x4fb335a84fa7d2d7
目标右半：0x02a183e3b4f4f49f

记 ROL64、ROR64 为 64 位循环移位，DELTA = 0x9e3779b97f4a7c15。对 i = 2, 3, ..., 31：
K[i] = ROL64(K[i-1] + K[i-2], 13)
       ^ ROR64(K[i-2], 7)
       ^ (i * DELTA)
先将 16 字节输入按小端拆成两个 64 位整数，并与前两个密钥异或
L = input_low64  ^ K[0]
R = input_high64 ^ K[1]
轮函数：
F(x, k) = (ROL64(x + k, 13) ^ ROL64(x, 29))
          + ROL64(x, 47)
之后使用 K[2]至 K[31]，共进行 30 轮
L, R = R, L ^ F(R, K[i])

对于Feistel 结构
正向：
new_L = old_R
new_R = old_L ^ F(old_R, K[i])
逆向：
old_R = new_L
old_L = new_R ^ F(new_L, K[i])
计算：
L, R = R ^ F(L, K[i]), L
再分别异或 K[0] 和 K[1]，按小端序打包为 16 字节，转成十六进制字符串
4865f2079724968a95c217e0ff4db6c6
```

import struct
from pathlib import Path
MASK64 = (1 << 64) - 1
DYNSYM = 0x3F8
def qword(mem, offset):
    return struct.unpack_from('<Q', mem, offset)[0]
def put_qword(mem, offset, value):
    struct.pack_into('<Q', mem, offset, value & MASK64)
def load_image(raw):
    if raw[:6] != b'\x7fELF\x02\x01':
        raise ValueError('expected an ELF64 little-endian image')
    phoff = struct.unpack_from('<Q', raw, 0x20)[0]
    phentsize, phnum = struct.unpack_from('<HH', raw, 0x36)
    loads = []
    for i in range(phnum):
        ph = struct.unpack_from('<IIQQQQQQ', raw, phoff + i * phentsize)
        if ph[0] == 1:  *# PT_LOAD*
            loads.append(ph)
    mem = bytearray(max(ph[3] + ph[6] for ph in loads) + 0x1000)
    for _, _, off, va, _, filesz, memsz, _ in loads:
        *# Linux maps complete pages, including file bytes past p_filesz.*
        file_start = off & ~0xFFF
        mem_start = va & ~0xFFF
        file_end = (off + filesz + 0xFFF) & ~0xFFF
        page_bytes = raw[file_start:file_end]
        mem[mem_start:mem_start + len(page_bytes)] = page_bytes
        if memsz > filesz:
            mem[va + filesz:va + memsz] = bytes(memsz - filesz)
    return mem
def recover_payload(raw):
    mem = load_image(raw)
    *# Simulate only the .rela.dyn operations needed by this sample.*
    *# Use load bias zero; imported C++ symbols are irrelevant here.*
    for i in range(25):
        off, info, addend = struct.unpack_from('<QQq', mem, 0x1388 + i * 24)
        symbol, kind = info >> 32, info & 0xFFFFFFFF
        if kind == 8:  *# R_X86_64_RELATIVE*
            put_qword(mem, off, addend)
        elif kind == 5 and symbol in (57, 49):  *# R_X86_64_COPY*
            entry = DYNSYM + symbol * 24
            src, size = qword(mem, entry + 8), qword(mem, entry + 16)
            mem[off:off + size] = bytes(mem[src:src + size])
    *# .rela.plt now contains the hidden relocation program.*
    for i in range(39):
        off, info, addend = struct.unpack_from('<QQq', mem, 0x15E0 + i * 24)
        symbol, kind = info >> 32, info & 0xFFFFFFFF
        entry = DYNSYM + symbol * 24
        if kind == 8:
            put_qword(mem, off, addend)
        elif kind == 33:  *# R_X86_64_SIZE64*
            put_qword(mem, off, qword(mem, entry + 16) + addend)
        elif kind == 1:  *# R_X86_64_64*
            if mem[entry + 4] & 0xF == 10:  *# STT_GNU_IFUNC*

                break  *# Recover its effect below, without calling native code.*
            put_qword(mem, off, qword(mem, entry + 8) + addend)
        else:

    else:

    stage1 = bytes(mem[0x7325:0x7396])
    acc = 0

def rol64(x, count):
    x &= MASK64
    return ((x << count) | (x >> (64 - count))) & MASK64
def ror64(x, count):
    return rol64(x, 64 - count)
def round_function(x, key):
    return ((rol64((x + key) & MASK64, 13) ^ rol64(x, 29))
            + rol64(x, 47)) & MASK64
def solve(mem):
    keys = [qword(mem, 0x7686) ^ qword(mem, 0x768E),
            qword(mem, 0x7696) ^ qword(mem, 0x769E)]
    for i in range(2, 32):
        keys.append(rol64((keys[-1] + keys[-2]) & MASK64, 13)
                    ^ ror64(keys[-2], 7)
                    ^ ((i * 0x9E3779B97F4A7C15) & MASK64))
    target = (qword(mem, 0x76A6), qword(mem, 0x76AE))
    left, right = target
    for i in range(31, 1, -1):
        left, right = right ^ round_function(left, keys[i]), left
    plain = struct.pack('<QQ', left ^ keys[0], right ^ keys[1])
    # Validate by running the recovered algorithm forward.
    left, right = struct.unpack('<QQ', plain)
    left ^= keys[0]
    right ^= keys[1]
    for i in range(2, 32):

    assert (left, right) == target
    return 'flag{' + plain.hex() + '}'
